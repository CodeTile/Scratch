# 🚀 2. FINAL Serilog-ready resilience extension

```csharp id="k8x2mc"
using System.Net;
using Microsoft.Extensions.Http.Resilience;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
using PollyTest.Configuration;

namespace PollyTest.Infrastructure.Http;

public static class MyApiResilienceExtensions
{
    public static IHttpClientBuilder AddMyApiResilience(
        this IHttpClientBuilder builder,
        IConfiguration config)
    {
        var options =
            config.GetSection("Resilience")
                   .Get<ResilienceOptions>()
            ?? new ResilienceOptions();

        builder.AddResilienceHandler("myapi-resilience", pipeline =>
        {
            // =====================================================
            // ⏱ TIMEOUT
            // =====================================================
            pipeline.AddTimeout(TimeSpan.FromSeconds(options.TimeoutSeconds));

            // =====================================================
            // 🔁 RETRY (Serilog logging)
            // =====================================================
            pipeline.AddRetry(new HttpRetryStrategyOptions
            {
                MaxRetryAttempts = options.Retry.MaxAttempts,
                Delay = TimeSpan.FromSeconds(options.Retry.DelaySeconds),
                BackoffType = DelayBackoffType.Exponential,
                UseJitter = true,

                ShouldHandle = args =>
                {
                    var response = args.Outcome.Result as HttpResponseMessage;

                    var method = response?.RequestMessage?.Method;

                    var retryableVerb =
                        method == HttpMethod.Get ||
                        method == HttpMethod.Put ||
                        method == HttpMethod.Delete;

                    var isTransient =
                        args.Outcome.Exception is HttpRequestException ||
                        response?.StatusCode is
                            HttpStatusCode.RequestTimeout or
                            HttpStatusCode.TooManyRequests or
                            HttpStatusCode.InternalServerError or
                            HttpStatusCode.BadGateway or
                            HttpStatusCode.ServiceUnavailable or
                            HttpStatusCode.GatewayTimeout;

                    return ValueTask.FromResult(retryableVerb && isTransient);
                },

                OnRetry = args =>
                {
                    var loggerFactory =
                        args.Context.ServiceProvider
                            .GetRequiredService<ILoggerFactory>();

                    var logger = loggerFactory.CreateLogger("MyApi.Resilience");

                    var response = args.Outcome.Result as HttpResponseMessage;

                    logger.LogWarning(
                        "HTTP Retry {Attempt} | {Method} {Url} | Status: {Status}",
                        args.AttemptNumber,
                        response?.RequestMessage?.Method,
                        response?.RequestMessage?.RequestUri,
                        response?.StatusCode
                    );

                    return ValueTask.CompletedTask;
                }
            });

            // =====================================================
            // 🔌 CIRCUIT BREAKER (Serilog logging)
            // =====================================================
            pipeline.AddCircuitBreaker(new HttpCircuitBreakerStrategyOptions
            {
                FailureRatio = options.CircuitBreaker.FailureRatio,
                MinimumThroughput = options.CircuitBreaker.MinimumThroughput,
                SamplingDuration = TimeSpan.FromSeconds(options.CircuitBreaker.SamplingDurationSeconds),
                BreakDuration = TimeSpan.FromSeconds(options.CircuitBreaker.BreakDurationSeconds),

                ShouldHandle = args =>
                {
                    var response = args.Outcome.Result as HttpResponseMessage;

                    return ValueTask.FromResult(
                        args.Outcome.Exception is HttpRequestException ||
                        response?.StatusCode is
                            HttpStatusCode.InternalServerError or
                            HttpStatusCode.BadGateway or
                            HttpStatusCode.ServiceUnavailable or
                            HttpStatusCode.GatewayTimeout);
                },

                OnOpened = args =>
                {
                    var loggerFactory =
                        args.Context.ServiceProvider
                            .GetRequiredService<ILoggerFactory>();

                    var logger = loggerFactory.CreateLogger("MyApi.Resilience");

                    logger.LogError(
                        "🚨 Circuit OPENED for {Endpoint}",
                        args.Context.OperationKey);

                    return ValueTask.CompletedTask;
                },

                OnClosed = args =>
                {
                    var loggerFactory =
                        args.Context.ServiceProvider
                            .GetRequiredService<ILoggerFactory>();

                    var logger = loggerFactory.CreateLogger("MyApi.Resilience");

                    logger.LogInformation(
                        "✅ Circuit CLOSED for {Endpoint}",
                        args.Context.OperationKey);

                    return ValueTask.CompletedTask;
                }
            });

            // =====================================================
            // 📡 FALLBACK (optional, Serilog-safe)
            // =====================================================
            pipeline.AddFallback(new HttpFallbackStrategyOptions<HttpResponseMessage>
            {
                ShouldHandle = args =>
                {
                    var response = args.Outcome.Result as HttpResponseMessage;

                    return ValueTask.FromResult(
                        response?.RequestMessage?.Method == HttpMethod.Get &&
                        (args.Outcome.Exception is HttpRequestException ||
                         response?.StatusCode >= HttpStatusCode.InternalServerError));
                },

                FallbackAction = args =>
                {
                    var loggerFactory =
                        args.Context.ServiceProvider
                            .GetRequiredService<ILoggerFactory>();

                    var logger = loggerFactory.CreateLogger("MyApi.Resilience");

                    logger.LogWarning(
                        "Fallback triggered for {Endpoint}",
                        args.Context.OperationKey);

                    var fallback = new HttpResponseMessage(HttpStatusCode.OK)
                    {
                        Content = new StringContent("""{"fallback":true}""")
                    };

                    return ValueTask.FromResult(new ValueTask<HttpResponseMessage>(fallback));
                }
            });
        });

        return builder;
    }
}
```
