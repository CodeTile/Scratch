using System.Linq.Expressions;
using Microsoft.EntityFrameworkCore;

public class GenericCrudRepository<TEntity>(DbContext context)
    : IGenericCrudRepository<TEntity> where TEntity : class
{
    private readonly DbContext _dbContext = context ?? throw new ArgumentNullException(nameof(context));

    public async Task<IServiceResult<TEntity>> AddAsync(
        TEntity entity,
        CancellationToken cancellationToken = default,
        params Expression<Func<TEntity, object>>[] references)
    {
        ArgumentNullException.ThrowIfNull(entity);

        _dbContext.Set<TEntity>().Add(entity);

        await LoadReferences(entity, references, cancellationToken);
        var result = await _dbContext.SaveChangesAsync(cancellationToken);

        return new ServiceResult<TEntity>
        {
            IsSuccess = result > 0,
            Value = entity
        };
    }

    public async Task<IServiceResult<IEnumerable<TEntity>>> AddBatchAsync(
        IEnumerable<TEntity> entities,
        CancellationToken cancellationToken = default,
        params Expression<Func<TEntity, object>>[] references)
    {
        ArgumentNullException.ThrowIfNull(entities);

        await _dbContext.Set<TEntity>().AddRangeAsync(entities, cancellationToken);

        await LoadReferences(entities, references, cancellationToken);
        var result = await _dbContext.SaveChangesAsync(cancellationToken);

        return new ServiceResult<IEnumerable<TEntity>>
        {
            IsSuccess = entities.Count() == result,
            Value = entities
        };
    }

    public async Task<IServiceResult<bool>> DeleteAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(predicate);

        var entities = FindEntities(predicate);
        if (entities == null || !await entities.AnyAsync(cancellationToken))
        {
            return ServiceResult<bool>.Failure(false, "Unable to find record to delete!");
        }

        _dbContext.RemoveRange(entities);

        try
        {
            int result = await _dbContext.SaveChangesAsync(cancellationToken);

            return result == 1
                ? ServiceResult<bool>.Success(true)
                : ServiceResult<bool>.Failure(false);
        }
        catch (Exception ex)
        {
            return ServiceResult<bool>.Failure(false, ConvertExceptionToUserMessage(ex));
        }
    }

    public IQueryable<TEntity>? FindEntities(
        Expression<Func<TEntity, bool>>? predicate = null,
        params string[] includes)
    {
        var query = ApplyIncludes(_dbContext.Set<TEntity>(), includes);
        return predicate != null ? query.Where(predicate) : query;
    }

    public async Task<TEntity?> GetByIdAsync(
        Expression<Func<TEntity, bool>> predicate,
        CancellationToken cancellationToken = default,
        params string[] includes)
    {
        ArgumentNullException.ThrowIfNull(predicate);

        var query = ApplyIncludes(_dbContext.Set<TEntity>(), includes);
        return await query.AsNoTracking().FirstOrDefaultAsync(predicate, cancellationToken);
    }

    public async Task<IServiceResult<TEntity>> UpdateAsync(
        TEntity entity,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(entity);

        _dbContext.Update(entity);

        return new ServiceResult<TEntity>
        {
            IsSuccess = (1 == await _dbContext.SaveChangesAsync(cancellationToken)),
            Value = entity
        };
    }

    public async Task<IServiceResult<IEnumerable<TEntity>>> UpdateAsync(
        IEnumerable<TEntity> entities,
        CancellationToken cancellationToken = default)
    {
        _dbContext.UpdateRange(entities);

        var result = await _dbContext.SaveChangesAsync(cancellationToken);

        return new ServiceResult<IEnumerable<TEntity>>
        {
            IsSuccess = entities.Count() == result,
            Value = entities
        };
    }

    internal async Task<bool> DeleteAsync(
        TEntity? entity,
        CancellationToken cancellationToken = default)
    {
        if (entity == null)
            return false;

        _dbContext.Set<TEntity>().Remove(entity);
        var result = await _dbContext.SaveChangesAsync(cancellationToken);

        return result == 1;
    }

    private static IQueryable<TEntity> ApplyIncludes(
        IQueryable<TEntity> query,
        IEnumerable<string> includes)
    {
        return includes.Aggregate(query, (current, include) => current.Include(include));
    }

    private static string ConvertExceptionToUserMessage(Exception ex)
    {
        ArgumentNullException.ThrowIfNull(ex);

        string message = ex.InnerException != null
            ? ConvertExceptionToUserMessage(ex.InnerException)
            : ex.Message;

        if (message.Contains("REFERENCE constraint", StringComparison.OrdinalIgnoreCase))
            return "Unable to delete, record is referenced elsewhere!";

        return message;
    }

    private async Task LoadReferences(
        TEntity entity,
        IEnumerable<Expression<Func<TEntity, object>>> references,
        CancellationToken cancellationToken)
    {
        foreach (var reference in references)
        {
            await _dbContext.Entry(entity).Reference(reference!).LoadAsync(cancellationToken);
        }
    }

    private async Task LoadReferences(
        IEnumerable<TEntity> entities,
        IEnumerable<Expression<Func<TEntity, object>>> references,
        CancellationToken cancellationToken)
    {
        foreach (var entity in entities)
        {
            await LoadReferences(entity, references, cancellationToken);
        }
    }
}

using Microsoft.EntityFrameworkCore;

    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.Add(new FooBar { Id = 1, Name = "Old" });
        await ctx.SaveChangesAsync();
        ctx.ChangeTracker.Clear();
        var repo = new GenericCrudRepository<FooBar>(ctx);

        // Act
        var result = await repo.UpdateAsync(updated, CancellationToken.None);

    }

    [Fact]
    public async Task UpdateAsync_ShouldThrow_WhenEntityNull()
    {
        // Arrange
        var ctx = CreateContext();
        var repo = new GenericCrudRepository<FooBar>(ctx);

        // Act + Assert
        await Should.ThrowAsync<ArgumentNullException>(() =>
            repo.UpdateAsync((FooBar)null!, CancellationToken.None));
    }

    [Fact]
    public async Task UpdateBatchAsync_ShouldUpdateEntities_AndReturnSuccess()
    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.AddRange(
            new FooBar { Id = 1, Name = "A" },
            new FooBar { Id = 2, Name = "B" }
        );
        await ctx.SaveChangesAsync();
        ctx.ChangeTracker.Clear();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var updated = new[]
        {
            new FooBar { Id = 1, Name = "X" },
            new FooBar { Id = 2, Name = "Y" }
        };

        // Act
        var result = await repo.UpdateAsync(updated, CancellationToken.None);

        // Assert
        result.IsSuccess.ShouldBeTrue();
        ctx.FooBars.First(e => e.Id == 1).Name.ShouldBe("X");
        ctx.FooBars.First(e => e.Id == 2).Name.ShouldBe("Y");
    }

using Microsoft.EntityFrameworkCore;
using Shouldly;
using Xunit;
using System.Threading;
using System.Linq.Expressions;

public class HelperTests
{
    private TestDbContext CreateContext()
    {
        var options = new DbContextOptionsBuilder<TestDbContext>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString())
            .Options;

        return new TestDbContext(options);
    }

    [Fact]
    public void ConvertExceptionToUserMessage_ShouldReturnReferenceMessage()
    {
        // Arrange
        var ex = new Exception("REFERENCE constraint failed");

        // Act
        var msg = typeof(GenericCrudRepository<FooBar>)
            .GetMethod("ConvertExceptionToUserMessage", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static)!
            .Invoke(null, new object[] { ex }) as string;

        // Assert
        msg.ShouldBe("Unable to delete, record is referenced elsewhere!");
    }

    [Fact]
    public void ConvertExceptionToUserMessage_ShouldReturnInnerMessage()
    {
        // Arrange
        var ex = new Exception("Outer", new Exception("Inner error"));

        // Act
        var msg = typeof(GenericCrudRepository<FooBar>)
            .GetMethod("ConvertExceptionToUserMessage", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static)!
            .Invoke(null, new object[] { ex }) as string;

        // Assert
        msg.ShouldBe("Inner error");
    }

    [Fact]
    public async Task LoadReferences_ShouldNotThrow()
    {
        // Arrange
        var ctx = CreateContext();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var entity = new FooBar { Id = 1 };

        var method = typeof(GenericCrudRepository<FooBar>)
            .GetMethod(
                "LoadReferences",
                System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance,
                null,
                new[] { typeof(FooBar), typeof(IEnumerable<Expression<Func<FooBar, object>>>), typeof(CancellationToken) },
                null
            )!;

        // Act + Assert
        await Should.NotThrowAsync(async () =>
            await (Task)method.Invoke(
                repo,
                new object[]
                {
                    entity,
                    Array.Empty<Expression<Func<FooBar, object>>>(),
                    CancellationToken.None
                }
            )!
        );
    }
}
