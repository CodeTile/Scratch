using System.Net.Http.Json;

public class MyClass(IHttpClientFactory httpClientFactory) : IMyClass
{
    private readonly HttpClient _httpClient = httpClientFactory.CreateClient("Hello");

    // ---------------- PUBLIC API ----------------

    public Task<string?> GetFooAsync(CancellationToken cancellationToken = default)
        => GetResultAsString("Foo", cancellationToken);

    public Task<string?> GetBarAsync(CancellationToken cancellationToken = default)
        => GetResultAsync<string>("Bar", cancellationToken);

    public Task<string?> PostFooAsync(object payload, CancellationToken cancellationToken = default)
        => PostResultAsString("Foo", payload, cancellationToken);

    public Task<TResponse?> PostBarAsync<TRequest, TResponse>(TRequest payload, CancellationToken cancellationToken = default)
        => PostResultAsync<TRequest, TResponse>("Bar", payload, cancellationToken);

    public Task<string?> PutAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PutResultAsString(url, payload, cancellationToken);

    public Task<TResponse?> PutAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PutResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<string?> PatchAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PatchResultAsString(url, payload, cancellationToken);

    public Task<TResponse?> PatchAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PatchResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<bool> DeleteAsync(string url, CancellationToken cancellationToken = default)
        => DeleteResult(url, cancellationToken);

    // ---------------- GET ----------------

    private Task<string?> GetResultAsString(string url, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.GetAsync(url, ct);
                if (!response.IsSuccessStatusCode)
                    return null;

                return await response.Content.ReadAsStringAsync(ct);
            },
            cancellationToken);

    private Task<T?> GetResultAsync<T>(string url, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.GetAsync(url, ct);
                if (!response.IsSuccessStatusCode)
                    return default;

                return await response.Content.ReadFromJsonAsync<T>(cancellationToken: ct);
            },
            cancellationToken);

    // ---------------- POST ----------------

    private Task<string?> PostResultAsString(string url, object payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.PostAsJsonAsync(url, payload, ct);
                if (!response.IsSuccessStatusCode)
                    return null;

                return await response.Content.ReadAsStringAsync(ct);
            },
            cancellationToken);

    private Task<TResponse?> PostResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.PostAsJsonAsync(url, payload, ct);
                if (!response.IsSuccessStatusCode)
                    return default;

                return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: ct);
            },
            cancellationToken);

    // ---------------- PUT ----------------

    private Task<string?> PutResultAsString(string url, object payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.PutAsJsonAsync(url, payload, ct);
                if (!response.IsSuccessStatusCode)
                    return null;

                return await response.Content.ReadAsStringAsync(ct);
            },
            cancellationToken);

    private Task<TResponse?> PutResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.PutAsJsonAsync(url, payload, ct);
                if (!response.IsSuccessStatusCode)
                    return default;

                return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: ct);
            },
            cancellationToken);

    // ---------------- PATCH ----------------

    private Task<string?> PatchResultAsString(string url, object payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var request = new HttpRequestMessage(HttpMethod.Patch, url)
                {
                    Content = JsonContent.Create(payload)
                };

                var response = await _httpClient.SendAsync(request, ct);
                if (!response.IsSuccessStatusCode)
                    return null;

                return await response.Content.ReadAsStringAsync(ct);
            },
            cancellationToken);

    private Task<TResponse?> PatchResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var request = new HttpRequestMessage(HttpMethod.Patch, url)
                {
                    Content = JsonContent.Create(payload)
                };

                var response = await _httpClient.SendAsync(request, ct);
                if (!response.IsSuccessStatusCode)
                    return default;

                return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: ct);
            },
            cancellationToken);

    // ---------------- DELETE ----------------

    private Task<bool> DeleteResult(string url, CancellationToken cancellationToken)
        => RetryAsync(
            async ct =>
            {
                var response = await _httpClient.DeleteAsync(url, ct);
                return response.IsSuccessStatusCode;
            },
            cancellationToken);

    // ---------------- RETRY ----------------

    private static async Task<T?> RetryAsync<T>(
        Func<CancellationToken, Task<T?>> action,
        CancellationToken cancellationToken,
        int retries = 2,
        int delayMs = 100)
    {
        for (int attempt = 0; attempt < retries; attempt++)
        {
            try
            {
                return await action(cancellationToken);
            }
            catch (HttpRequestException) when (attempt < retries - 1)
            {
                await Task.Delay(delayMs, cancellationToken);
            }
            catch (TaskCanceledException) when (!cancellationToken.IsCancellationRequested && attempt < retries - 1)
            {
                await Task.Delay(delayMs, cancellationToken);
            }
        }

        return await action(cancellationToken);
    }
}
