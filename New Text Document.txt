using System.Net.Http.Json;

namespace MyApp;

public class MyClass(IHttpClientFactory httpClientFactory) : IMyClass
{
    private readonly HttpClient _httpClient = httpClientFactory.CreateClient("MyClient");

    public Task<string?> GetFooAsync(CancellationToken cancellationToken = default)
        => GetResultAsString("Foo", cancellationToken);

    public Task<string?> GetBarAsync(CancellationToken cancellationToken = default)
        => GetResultAsync<string>("Bar", cancellationToken);

    public Task<string?> PostFooAsync(object payload, CancellationToken cancellationToken = default)
        => PostResultAsString("Foo", payload, cancellationToken);

    public Task<TResponse?> PostBarAsync<TRequest, TResponse>(TRequest payload, CancellationToken cancellationToken = default)
        => PostResultAsync<TRequest, TResponse>("Bar", payload, cancellationToken);

    public Task<string?> PutAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PutResultAsString(url, payload, cancellationToken);

    public Task<TResponse?> PutAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PutResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<string?> PatchAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PatchResultAsString(url, payload, cancellationToken);

    public Task<TResponse?> PatchAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PatchResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<bool> DeleteAsync(string url, CancellationToken cancellationToken = default)
        => DeleteResult(url, cancellationToken);

    // ---------------- GET ----------------

    private async Task<string?> GetResultAsString(string url, CancellationToken cancellationToken)
    {
        var response = await _httpClient.GetAsync(url, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<T?> GetResultAsync<T>(string url, CancellationToken cancellationToken)
    {
        var response = await _httpClient.GetAsync(url, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<T>(cancellationToken: cancellationToken);
    }

    // ---------------- POST ----------------

    private async Task<string?> PostResultAsString(string url, object payload, CancellationToken cancellationToken)
    {
        var response = await _httpClient.PostAsJsonAsync(url, payload, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PostResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        var response = await _httpClient.PostAsJsonAsync(url, payload, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- PUT ----------------

    private async Task<string?> PutResultAsString(string url, object payload, CancellationToken cancellationToken)
    {
        var response = await _httpClient.PutAsJsonAsync(url, payload, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PutResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        var response = await _httpClient.PutAsJsonAsync(url, payload, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- PATCH ----------------

    private async Task<string?> PatchResultAsString(string url, object payload, CancellationToken cancellationToken)
    {
        using var request = new HttpRequestMessage(HttpMethod.Patch, url)
        {
            Content = JsonContent.Create(payload)
        };

        var response = await _httpClient.SendAsync(request, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PatchResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        using var request = new HttpRequestMessage(HttpMethod.Patch, url)
        {
            Content = JsonContent.Create(payload)
        };

        var response = await _httpClient.SendAsync(request, cancellationToken);
        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- DELETE ----------------

    private async Task<bool> DeleteResult(string url, CancellationToken cancellationToken)
    {
        var response = await _httpClient.DeleteAsync(url, cancellationToken);
        return response.IsSuccessStatusCode;
    }
}

using System.Net;
using System.Net.Http.Json;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Options;
using Moq;
using Polly.Timeout;
using Shouldly;
using Xunit;

namespace MyApp.Tests;

public class MyClassTests

    private static readonly PollySettings DefaultSettings = new()
    {
        RetryCount = 3,
        RetryBaseDelayMs = 1,
        TimeoutSeconds = 1

    // ---------------------------------------------------------
    // GET
    // ---------------------------------------------------------

    [Fact]
    public async Task GetFooAsync_ReturnsString_OnSuccess()
    {
        // Arrange
        var handler = new FakeHandler((_, _) =>
            Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new StringContent("hello")
            }));
        var sut = CreateSut(handler);

        // Act
        var result = await sut.GetFooAsync();

        // Assert
        result.ShouldBe("hello");
    }

    [Fact]
    public async Task GetBarAsync_ReturnsDeserialized_OnSuccess()
    {
        // Arrange
        var handler = new FakeHandler((_, _) =>
        {
            var response = new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = JsonContent.Create(new TestDto("value"))
            };
            return Task.FromResult(response);
        });
        var sut = CreateSut(handler);

        // Act
        var result = await sut.GetBarAsync();

        // Assert
        result.ShouldBe("value");
    }

    // ---------------------------------------------------------
    // POST
    // ---------------------------------------------------------

    [Fact]
    public async Task PostFooAsync_ReturnsNull_OnNonSuccess()
    {
        // Arrange
        var handler = new FakeHandler((_, _) =>
            Task.FromResult(new HttpResponseMessage(HttpStatusCode.BadRequest)));
        var sut = CreateSut(handler);

        // Act
        var result = await sut.PostFooAsync(new { Name = "PingPong" });

        // Assert
        result.ShouldBeNull();
    }

    [Fact]
    public async Task PostBarAsync_Retries_OnTransientStatus_ThenSucceeds()
    {
        // Arrange
        var calls = 0;
        var handler = new FakeHandler((_, _) =>
        {
            calls++;
            if (calls == 1)
                return Task.FromResult(new HttpResponseMessage(HttpStatusCode.InternalServerError));

            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = JsonContent.Create(new TestDto("ok"))
            });
        });

        var sut = CreateSut(handler);

        // Act
        var result = await sut.PostBarAsync<object, TestDto>(new { Name = "PingPong" });

        // Assert
        result!.Value.ShouldBe("ok");
        calls.ShouldBe(2);
    }

    // ---------------------------------------------------------
    // PUT

    [Fact]
    public async Task PutAsync_ReturnsString_OnSuccess()

    {
        // Arrange
        var calls = 0;
        var handler = new FakeHandler((_, _) =>
        {
            calls++;
            if (calls == 1)
                throw new HttpRequestException("boom");

            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new StringContent("ok")
            });

        var sut = CreateSut(handler);

        // Act
        var result = await sut.GetFooAsync();

        var calls = 0;
        var handler = new FakeHandler((_, _) =>

            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.BadRequest));

        var sut = CreateSut(handler);

    // TIMEOUT & CANCELLATION

    [Fact]
    public async Task ThrowsTimeoutRejectedException_OnTimeout()
    {
        // Arrange
        var handler = new FakeHandler(async (_, ct) =>
        {
            await Task.Delay(TimeSpan.FromSeconds(5), ct);
            return new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new StringContent("late")
            };
        });

        var sut = CreateSut(handler);

        // Act
        var act = () => sut.GetFooAsync();

        // Assert
        await act.ShouldThrowAsync<TimeoutRejectedException>();
    }

    [Fact]
    public async Task Respects_ExternalCancellationToken()
    {
        // Arrange
        var handler = new FakeHandler(async (_, ct) =>
        {
            await Task.Delay(TimeSpan.FromSeconds(5), ct);
            return new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new StringContent("never")
            };
        });

        var sut = CreateSut(handler);
        using var cts = new CancellationTokenSource();
        cts.CancelAfter(TimeSpan.FromMilliseconds(50));

        // Act
        var act = () => sut.GetFooAsync(cts.Token);

        // Assert
        await act.ShouldThrowAsync<TaskCanceledException>();
    }

    // ---------------------------------------------------------
    // TEST HELPERS
    // ---------------------------------------------------------

    private static MyClass CreateSut(HttpMessageHandler handler)
    {
        var services = new ServiceCollection();

        // Register Polly settings
        services.Configure<PollySettings>(opts =>
        {
            opts.RetryCount = DefaultSettings.RetryCount;
            opts.RetryBaseDelayMs = DefaultSettings.RetryBaseDelayMs;
            opts.TimeoutSeconds = DefaultSettings.TimeoutSeconds;
        });

        // Register HttpClient with handler
        services.AddHttpClient("MyClient")
            .ConfigurePrimaryHttpMessageHandler(() => handler);

        // Register MyClass
        services.AddTransient<IMyClass, MyClass>();

        var provider = services.BuildServiceProvider();

        return provider.GetRequiredService<IMyClass>();
    }

    private sealed class FakeHandler : HttpMessageHandler
    {
        private readonly Func<HttpRequestMessage, CancellationToken, Task<HttpResponseMessage>> _handler;

        public FakeHandler(Func<HttpRequestMessage, CancellationToken, Task<HttpResponseMessage>> handler)
        {
            _handler = handler;
        }

        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
            => _handler(request, cancellationToken);
    }

    private readonly record struct TestDto(string Value);
}
