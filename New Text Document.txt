using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using System.Linq.Expressions;
using System.Reflection;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.ChangeTracking;
using Moq;
using Xunit;

// ------------------------------------------------------------
// Entity under test
// ------------------------------------------------------------
public class Foobar
{
    public int Id { get; set; }
    public string? Foo { get; set; }
    public int Bar { get; set; }
}

// ------------------------------------------------------------
// TestQueryable<T> to intercept Include() calls
// ------------------------------------------------------------
public class TestQueryable<T> : IQueryable<T>
{
    private readonly IQueryable<T> _inner;

    public TestQueryable(IQueryable<T> inner, IQueryProvider provider)
    {
        _inner = inner;
        Provider = provider;
        Expression = inner.Expression;
    }

    public IEnumerator<T> GetEnumerator() => _inner.GetEnumerator();
    IEnumerator IEnumerable.GetEnumerator() => GetEnumerator();

    public Type ElementType => typeof(T);
    public Expression Expression { get; }
    public IQueryProvider Provider { get; }
}

// ------------------------------------------------------------
// Reflection helpers to call private methods
// ------------------------------------------------------------
public static class RepoTestExtensions
{
    public static IQueryable<T> InvokeApplyIncludes<T>(
        IEnumerable<string> includes,
        IQueryable<T> query)
    {
        var method = typeof(GenericCrudRepository<T>)
            .GetMethod("ApplyIncludes", BindingFlags.NonPublic | BindingFlags.Static);

        return (IQueryable<T>)method.Invoke(null, new object[] { query, includes });
    }

    public static Task InvokeLoadReferences<TEntity>(
        this GenericCrudRepository<TEntity> repo,
        TEntity entity,
        IEnumerable<Expression<Func<TEntity, object>>> refs)
        where TEntity : class
    {
        var method = typeof(GenericCrudRepository<TEntity>)
            .GetMethod("LoadReferences", BindingFlags.NonPublic | BindingFlags.Instance,
                null,
                new[] { typeof(TEntity), typeof(IEnumerable<Expression<Func<TEntity, object>>>) },
                null);

        return (Task)method.Invoke(repo, new object[] { entity, refs });
    }

    public static Task InvokeLoadReferences<TEntity>(
        this GenericCrudRepository<TEntity> repo,
        IEnumerable<TEntity> entities,
        IEnumerable<Expression<Func<TEntity, object>>> refs)
        where TEntity : class
    {
        var method = typeof(GenericCrudRepository<TEntity>)
            .GetMethod("LoadReferences", BindingFlags.NonPublic | BindingFlags.Instance,
                null,
                new[] { typeof(IEnumerable<TEntity>), typeof(IEnumerable<Expression<Func<TEntity, object>>>) },
                null);

        return (Task)method.Invoke(repo, new object[] { entities, refs });
    }
}

// ------------------------------------------------------------
// TESTS (Moq version)
// ------------------------------------------------------------
public class GenericCrudRepositoryTests
{
    // --------------------------------------------------------
    // 1. ApplyIncludes
    // --------------------------------------------------------
    [Fact]
    public void ApplyIncludes_AppliesAllIncludes()
    {
        // Arrange
        var data = new List<Foobar>().AsQueryable();

        var provider = new Mock<IQueryProvider>();
        var queryable = new TestQueryable<Foobar>(data, provider.Object);

        var includes = new[] { "Foo", "Bar", "SomethingElse" };

        // Act
        var result = RepoTestExtensions.InvokeApplyIncludes(includes, queryable);

        // Assert
        provider.Verify(p =>
            p.CreateQuery(It.Is<MethodCallExpression>(expr =>
                expr.Method.Name == "Include")),
            Times.Exactly(includes.Length));
    }

    // --------------------------------------------------------
    // 2. LoadReferences (single entity)
    // --------------------------------------------------------
    [Fact]
    public async Task LoadReferences_LoadsEachReference()
    {
        // Arrange
        var entity = new Foobar();

        var mockContext = new Mock<DbContext>();
        var mockEntry = new Mock<EntityEntry<Foobar>>();
        var mockRef = new Mock<ReferenceEntry<Foobar, object>>();

        mockContext.Setup(c => c.Entry(entity)).Returns(mockEntry.Object);
        mockEntry.Setup(e => e.Reference(It.IsAny<Expression<Func<Foobar, object>>>()))
                 .Returns(mockRef.Object);

        var repo = new GenericCrudRepository<Foobar>(mockContext.Object);

        Expression<Func<Foobar, object>> r1 = x => x.Foo!;
        Expression<Func<Foobar, object>> r2 = x => x.Bar;

        // Act
        await repo.InvokeLoadReferences(entity, new[] { r1, r2 });

        // Assert
        mockRef.Verify(r => r.LoadAsync(default), Times.Exactly(2));
    }

    // --------------------------------------------------------
    // 3. LoadReferences (multiple entities)
    // --------------------------------------------------------
    [Fact]
    public async Task LoadReferences_ForMultipleEntities_LoadsReferencesForEach()
    {
        // Arrange
        var e1 = new Foobar();
        var e2 = new Foobar();

        var mockContext = new Mock<DbContext>();
        var mockEntry = new Mock<EntityEntry<Foobar>>();
        var mockRef = new Mock<ReferenceEntry<Foobar, object>>();

        mockContext.Setup(c => c.Entry(It.IsAny<Foobar>())).Returns(mockEntry.Object);
        mockEntry.Setup(e => e.Reference(It.IsAny<Expression<Func<Foobar, object>>>()))
                 .Returns(mockRef.Object);

        var repo = new GenericCrudRepository<Foobar>(mockContext.Object);

        Expression<Func<Foobar, object>> r1 = x => x.Foo!;

        // Act
        await repo.InvokeLoadReferences(new[] { e1, e2 }, new[] { r1 });

        // Assert
        mockRef.Verify(r => r.LoadAsync(default), Times.Exactly(2));
    }
}
