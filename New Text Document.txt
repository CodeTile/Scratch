# 🚀 4. AD service with BEST caching strategy

This is the important part.

✔ Sliding expiration = keeps active users fast
✔ Absolute expiration = ensures fresh AD data eventually

```csharp id="m9q2lk"
using System.DirectoryServices;
using System.DirectoryServices.AccountManagement;
using Microsoft.Extensions.Caching.Memory;

public class AdService
{
    private readonly IMemoryCache _cache;

    public AdService(IMemoryCache cache)
    {
        _cache = cache;
    }

    public AdUserInfo? GetUserInfo(string username)
    {
        // Normalize DOMAIN\user → user
        var sam = username.Contains("\\")
            ? username.Split('\\')[1]
            : username;

        var cacheKey = $"ad_user_{sam}";

        return _cache.GetOrCreate(cacheKey, entry =>
        {
            // ⭐ BEST PRACTICE CACHE POLICY
            entry.SlidingExpiration = TimeSpan.FromMinutes(3);
            entry.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10);

            using var context = new PrincipalContext(ContextType.Domain);
            var user = UserPrincipal.FindByIdentity(context, sam);

            if (user == null)
                return null;

            var de = (DirectoryEntry)user.GetUnderlyingObject();

            return new AdUserInfo
            {
                Email = de.Properties["mail"]?.Value?.ToString(),
                Department = de.Properties["department"]?.Value?.ToString(),
                JobTitle = de.Properties["title"]?.Value?.ToString()
            };
        });
    }
}
```

# 🔐 5. Claims transformation (inject AD data into user)

This makes AD data available everywhere in Blazor as claims.

```csharp id="v0h8dd"
using System.Security.Claims;
using Microsoft.AspNetCore.Authentication;

public class AdClaimsTransformation : IClaimsTransformation
{
    private readonly AdService _adService;

    public AdClaimsTransformation(AdService adService)
    {
        _adService = adService;
    }

    public Task<ClaimsPrincipal> TransformAsync(ClaimsPrincipal principal)
    {
        var identity = (ClaimsIdentity)principal.Identity!;

        // Avoid duplicate work
        if (identity.HasClaim(c => c.Type == "department"))
            return Task.FromResult(principal);

        var username = principal.Identity?.Name;

        if (!string.IsNullOrEmpty(username))
        {
            var info = _adService.GetUserInfo(username);

            if (info != null)
            {
                if (!string.IsNullOrEmpty(info.Email))
                    identity.AddClaim(new Claim(ClaimTypes.Email, info.Email));

                if (!string.IsNullOrEmpty(info.Department))
                    identity.AddClaim(new Claim("department", info.Department));

                if (!string.IsNullOrEmpty(info.JobTitle))
                    identity.AddClaim(new Claim("jobTitle", info.JobTitle));
            }
        }

        return Task.FromResult(principal);
    }
}
```

## ✔ Test: caching prevents second AD call

[Fact]
public void GetUserInfo_UsesCache_NotADTwice()
{
    var cache = new MemoryCache(new MemoryCacheOptions());

    var adMock = new Mock<IAdDirectory>();

    adMock.Setup(x => x.GetUser(It.IsAny<string>()))
        .Returns(new AdUserInfo { Email = "a@b.com" });

    var service = new AdService(cache, adMock.Object);

    // First call → hits AD
    var r1 = service.GetUserInfo("DOMAIN\\jdoe");

    // Second call → should NOT hit AD again
    var r2 = service.GetUserInfo("DOMAIN\\jdoe");

    adMock.Verify(x => x.GetUser("DOMAIN\\jdoe"), Times.Once);

    Assert.Equal(r1!.Email, r2!.Email);
}

using System.Security.Claims;
using Xunit;
using Moq;

public class ClaimsTests
{
    [Fact]
    public async Task AddsAdClaims()
    {
        var cache = new MemoryCache(new MemoryCacheOptions());

        var adMock = new Mock<IAdDirectory>();
        adMock.Setup(x => x.GetUser("DOMAIN\\jdoe"))
            .Returns(new AdUserInfo
            {
                Email = "jdoe@test.com",
                Department = "IT",
                JobTitle = "Dev"
            });

        var service = new AdService(cache, adMock.Object);
        var transformer = new AdClaimsTransformation(service);

        var identity = new ClaimsIdentity(new[]
        {
            new Claim(ClaimTypes.Name, "DOMAIN\\jdoe")
        }, "Test");

        var principal = new ClaimsPrincipal(identity);

        var result = await transformer.TransformAsync(principal);

        Assert.Contains(result.Claims, c => c.Type == ClaimTypes.Email);
        Assert.Contains(result.Claims, c => c.Type == "department");
        Assert.Contains(result.Claims, c => c.Type == "jobTitle");
    }
}
