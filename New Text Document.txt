# 🎯 1. Example Controller (for context)

Let’s assume you have a controller like this:

```csharp
[ApiController]
[Route("api/[controller]")]
public class WeatherController : ControllerBase
{
    private readonly IWeatherService _service;

    public WeatherController(IWeatherService service)
    {
        _service = service;
    }

    [HttpGet("{city}")]
    public async Task<IActionResult> GetWeather(string city)
    {
        if (string.IsNullOrWhiteSpace(city))
            return BadRequest("City is required");

        var result = await _service.GetWeatherAsync(city);

        if (result == null)
            return NotFound();

        return Ok(result);
    }
}
```

# 🧪 3. Unit Test Class

```csharp
using Xunit;
using Moq;
using Microsoft.AspNetCore.Mvc;

public class WeatherControllerTests
{
    private readonly Mock<IWeatherService> _serviceMock;
    private readonly WeatherController _controller;

    public WeatherControllerTests()
    {
        _serviceMock = new Mock<IWeatherService>();
        _controller = new WeatherController(_serviceMock.Object);
    }

    [Fact]
    public async Task GetWeather_ReturnsBadRequest_WhenCityIsEmpty()
    {
        // Act
        var result = await _controller.GetWeather("");

        // Assert
        var badRequest = Assert.IsType<BadRequestObjectResult>(result);
        Assert.Equal("City is required", badRequest.Value);
    }

    [Fact]
    public async Task GetWeather_ReturnsNotFound_WhenServiceReturnsNull()
    {
        // Arrange
        _serviceMock
            .Setup(s => s.GetWeatherAsync("London"))
            .ReturnsAsync((WeatherResult)null);

        // Act
        var result = await _controller.GetWeather("London");

        // Assert
        Assert.IsType<NotFoundResult>(result);
    }

    [Fact]
    public async Task GetWeather_ReturnsOk_WithWeatherData()
    {
        // Arrange
        var expected = new WeatherResult { Temperature = 20 };
        
        _serviceMock
            .Setup(s => s.GetWeatherAsync("London"))
            .ReturnsAsync(expected);

        // Act
        var result = await _controller.GetWeather("London");

        // Assert
        var ok = Assert.IsType<OkObjectResult>(result);
        Assert.Equal(expected, ok.Value);
    }

    [Fact]
    public async Task GetWeather_CallsServiceExactlyOnce()
    {
        // Arrange
        _serviceMock
            .Setup(s => s.GetWeatherAsync("London"))
            .ReturnsAsync(new WeatherResult());

        // Act
        await _controller.GetWeather("London");

        // Assert
        _serviceMock.Verify(s => s.GetWeatherAsync("London"), Times.Once);
    }
}
```
