## 4. Complete Authorization Engine with Domain Validation (AuthorizationEngine.cs)

using Microsoft.AspNetCore.Authorization;using Microsoft.AspNetCore.Http;using Microsoft.Extensions.Configuration;using Microsoft.Extensions.Logging;using Microsoft.Extensions.Options;using System.Security.Claims;
namespace YourApi.Authorization;
public enum ApiRole
{
    None = 0,
    Viewer = 10,
    Editor = 20,
    Auditor = 30,
    Manager = 40,
    ServiceDesk = 50,
    Developer = 60
}

[AttributeUsage(AttributeTargets.Method | AttributeTargets.Class, AllowMultiple = false, Inherited = true)]public class MinimumRoleAttribute : AuthorizeAttribute
{
    public const string PolicyPrefix = "MinimumRole:";

    public MinimumRoleAttribute(ApiRole role)
    {
        Policy = $"{PolicyPrefix}{role}";
    }
}
public class MinimumRoleRequirement(ApiRole minimumRole) : IAuthorizationRequirement
{
    public ApiRole MinimumRole { get; } = minimumRole;
}
public class DynamicRolePolicyProvider(IOptions<AuthorizationOptions> options) : DefaultAuthorizationPolicyProvider(options)
{
    public override async Task<AuthorizationPolicy?> GetPolicyAsync(string policyName)
    {
        if (policyName.StartsWith(MinimumRoleAttribute.PolicyPrefix, StringComparison.OrdinalIgnoreCase))
        {
            var roleStr = policyName[MinimumRoleAttribute.PolicyPrefix.Length..];
            
            if (Enum.TryParse<ApiRole>(roleStr, true, out var role))
            {
                var policy = new AuthorizationPolicyBuilder();
                policy.AddRequirements(new MinimumRoleRequirement(role));
                return policy.Build();
            }
        }

        return await base.GetPolicyAsync(policyName);
    }
}
public class MinimumRoleHandler : AuthorizationHandler<MinimumRoleRequirement>
{
    private readonly Dictionary<ApiRole, string[]> _roleGroupsLookup = new();
    private readonly string _allowedDomain;
    private readonly ILogger<MinimumRoleHandler> _logger;
    private readonly IHttpContextAccessor _httpContextAccessor;

    public MinimumRoleHandler(
        IConfiguration configuration, 
        ILogger<MinimumRoleHandler> logger,
        IHttpContextAccessor httpContextAccessor)
    {
        _logger = logger;
        _httpContextAccessor = httpContextAccessor;
        _allowedDomain = configuration["AllowedDomain"] ?? string.Empty;

        foreach (ApiRole role in Enum.GetValues<ApiRole>())
        {
            string? rawAdGroups = configuration[$"RoleMappings:{role}"];
            if (!string.IsNullOrEmpty(rawAdGroups))
            {
                string[] cleanGroups = rawAdGroups.Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries);
                _roleGroupsLookup[role] = cleanGroups;
            }
        }
    }

    protected override Task HandleRequirementAsync(AuthorizationHandlerContext context, MinimumRoleRequirement requirement)
    {
        var username = context.User.Identity?.Name ?? "Anonymous";

        if (context.User.Identity is not { IsAuthenticated: true })
        {
            _logger.LogWarning("Access Denied: Unauthenticated access attempt by {Username}.", username);
            return Task.CompletedTask;
        }

        // Domain Filtering Validation Check
        if (!string.IsNullOrEmpty(_allowedDomain) && username.Contains('\\'))
        {
            var userDomain = username.Split('\\')[0];
            if (!string.Equals(userDomain, _allowedDomain, StringComparison.OrdinalIgnoreCase))
            {
                _logger.LogCritical("Security Alert: User '{Username}' from an untrusted domain '{UserDomain}' attempted access. Expected domain was '{AllowedDomain}'.", 
                    username, userDomain, _allowedDomain);
                
                LogAccessDenied(context, requirement, username, "Untrusted Domain Source Context");
                return Task.CompletedTask; // Abort instantly without verifying any AD group credentials
            }
        }

        foreach (ApiRole role in Enum.GetValues<ApiRole>())
        {
            if (role >= requirement.MinimumRole)
            {
                if (_roleGroupsLookup.TryGetValue(role, out var adGroups))
                {
                    foreach (var group in adGroups)
                    {
                        if (context.User.IsInRole(group))
                        {
                            context.Succeed(requirement);
                            return Task.CompletedTask;
                        }
                    }
                }
            }
        }

        LogAccessDenied(context, requirement, username, "Insufficient Permissions / Missing Active Directory Group Membership");
        return Task.CompletedTask;
    }

    private void LogAccessDenied(AuthorizationHandlerContext context, MinimumRoleRequirement requirement, string username, string reason)
    {
        var httpContext = _httpContextAccessor.HttpContext;
        var requestPath = httpContext?.Request.Path.Value ?? "Unknown Path";
        var httpMethod = httpContext?.Request.Method ?? "UNKNOWN";

        var userAdGroups = context.User.Claims
            .Where(c => c.Type == ClaimTypes.GroupSid || c.Type == "http://microsoft.com")
            .Select(c => c.Value)
            .ToList();

        _logger.LogWarning(
            "Access Denied: User {Username} failed authorization rules for endpoint {HttpMethod} {RequestPath}. " +
            "Reason: {DenialReason}. " +
            "Required minimum role was {RequiredRole} ({RequiredValue}). " +
            "Assigned User SIDs: {UserGroupSIDs}",
            username,
            httpMethod,
            requestPath,
            reason,
            requirement.MinimumRole,
            (int)requirement.MinimumRole,
            userAdGroups
        );
    }
}

## 5. Application Startup (Program.cs)

using Microsoft.AspNetCore.Authentication.Negotiate;using Microsoft.AspNetCore.Authorization;using Microsoft.OpenApi.Models;using Serilog;using YourApi.Authorization;
var builder = WebApplication.CreateBuilder(args);

Log.Logger = new LoggerConfiguration()
    .ReadFrom.Configuration(builder.Configuration)
    .CreateLogger();

builder.Host.UseSerilog();

builder.Services.AddAuthentication(NegotiateDefaults.AuthenticationScheme)
    .AddNegotiate();
// .NET 10 Native OpenAPI Document Generation Configuration
builder.Services.AddOpenApi(options =>
{
    options.AddDocumentTransformer((document, context, cancellationToken) =>
    {
        document.Info = new OpenApiInfo
        {
            Title = "Secure Windows Authentication API System",
            Version = "v1",
            Description = "Corporate Intranet Services API Engine with Hierarchical Role Controls"
        };

        // Inject Native Windows HTTP Security Schemes into the OpenAPI specification
        document.Components ??= new OpenApiComponents();
        document.Components.SecuritySchemes.Add("WindowsAuth", new OpenApiSecurityScheme
        {
            Type = SecuritySchemeType.Http,
            Scheme = "negotiate",
            Description = "Natively prompts for your local corporate Windows Domain OS profile credentials automatically."
        });

        document.SecurityRequirements.Add(new OpenApiSecurityRequirement
        {
            {
                new OpenApiSecurityScheme
                {
                    Reference = new OpenApiReference { Type = ReferenceType.SecurityScheme, Id = "WindowsAuth" }
                },
                Array.Empty<string>()
            }
        });

        return Task.CompletedTask;
    });
});

builder.Services.AddHttpContextAccessor();
builder.Services.AddSingleton<IAuthorizationPolicyProvider, DynamicRolePolicyProvider>();
builder.Services.AddScoped<IAuthorizationHandler, MinimumRoleHandler>();

builder.Services.AddControllers();
var app = builder.Build();

app.UseSerilogRequestLogging();
if (app.Environment.IsDevelopment())
{
    // Map the Native OpenAPI spec generation endpoint route
    app.MapOpenApi();
    
    // Serve up the interactive Swagger UI interface sourced from the native OpenAPI document configuration
    app.UseSwaggerUI(options =>
    {
        options.SwaggerEndpoint("/openapi/v1.json", "Secure API Systems v1 Specification Metadata");
        options.RoutePrefix = "swagger";
    });
}

app.UseAuthentication();
app.UseAuthorization();

app.MapControllers();
try
{
    Log.Information("Starting API Engine Host...");
    app.Run();
}catch (Exception ex)
{
    Log.Fatal(ex, "API Engine Host terminated unexpectedly");
}finally
{
    Log.CloseAndFlush();
}

To obscure the existence of secure endpoints from unauthorized users, you can intercept the authorization failure and rewrite the standard 403 Forbidden or 401 Unauthorized responses into a standard 404 Not Found response. [1, 2] 
In ASP.NET Core, the cleanest way to do this without bypassing the native authorization pipelines is by creating a custom Authorization Middleware Result Handler. This intercepts the decision after your MinimumRoleHandler finishes evaluation, checks if the user failed validation, and swaps the response payload on the fly.
Here are the code blocks ready to be cut and pasted into your project.
## 1. Create the Custom Middleware Result Handler (ForbiddenToNotFoundHandler.cs)
Create a new file named ForbiddenToNotFoundHandler.cs and paste this code. It safely converts any authorization failure into a clean 404 Not Found response sent to the client, while keeping your internal Serilog warnings fully accurate.

using Microsoft.AspNetCore.Authorization;using Microsoft.AspNetCore.Authorization.Policy;using Microsoft.AspNetCore.Http;
namespace YourApi.Authorization;
public class ForbiddenToNotFoundHandler : IAuthorizationMiddlewareResultHandler
{
    private readonly AuthorizationMiddlewareResultHandler _defaultHandler = new();

    public async Task HandleAsync(
        RequestDelegate next,
        HttpContext context,
        AuthorizationPolicy policy,
        PolicyAuthorizationResult authorizeResult)
    {
        // Check if the authorization evaluation failed (either unauthenticated or lacked required groups)
        if (authorizeResult.Challenged || authorizeResult.Forbidden)
        {
            // Override the response status to 404 Not Found
            context.Response.StatusCode = StatusCodes.Status404NotFound;
            
            // Optional: Provide a clean, generic JSON payload matching a standard resource lookup failure
            context.Response.ContentType = "application/json";
            await context.Response.WriteAsJsonAsync(new 
            { 
                Status = 404, 
                Error = "Not Found", 
                Message = "The requested resource could not be found on this server." 
            });
            
            return; // Short-circuit the request pipeline instantly
        }

        // If authorization succeeded, let the request flow through to the controller naturally
        await _defaultHandler.HandleAsync(next, context, policy, authorizeResult);
    }
}

## Why this approach is optimal:

   1. Perfect Obscurity: External malicious actors or unauthorized employees probing your internal API paths will receive a standard 404 Not Found response. They will have no idea whether the endpoint exists or if they simply lack permissions. [3] 
   2. Uncompromised Auditing: Your Serilog audit files will still log the specific details via the MinimumRoleHandler (e.g., "Access Denied: User DOMAIN\jdoe failed authorization rules for endpoint POST /api/data/purge"). Your internal administrators retain clear security visibility while keeping the outside client completely in the dark.
