using Microsoft.Extensions.Caching.Memory;

namespace YourApp.Security;

public interface IAdCache
{
    T? GetOrCreate<T>(string key, Func<T> factory);
}

public class AdCache : IAdCache
{
    private readonly IMemoryCache _cache;

    public AdCache(IMemoryCache cache)
    {
        _cache = cache;
    }

    public T? GetOrCreate<T>(string key, Func<T> factory)
    {
        return _cache.GetOrCreate(key, entry =>
        {
            entry.SlidingExpiration = TimeSpan.FromMinutes(7);
            entry.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10);

            return factory();
        });
    }
}

# 🧠 5. GROUP RESOLVER (NESTED SUPPORT)

```csharp
namespace YourApp.Security;

public interface IGroupResolver
{
    List<string> Expand(List<string> directGroups);
}

public class GroupResolver : IGroupResolver
{
    private readonly IDictionary<string, List<string>> _map;

    public GroupResolver(IDictionary<string, List<string>> map)
    {
        _map = map;
    }

    public List<string> Expand(List<string> directGroups)
    {
        var result = new HashSet<string>();
        var visited = new HashSet<string>();

        foreach (var g in directGroups)
            Resolve(g, result, visited);

        return result.ToList();
    }

    private void Resolve(string g, HashSet<string> result, HashSet<string> visited)
    {
        if (!visited.Add(g)) return;

        result.Add(g);

        if (_map.TryGetValue(g, out var children))
        {
            foreach (var c in children)
                Resolve(c, result, visited);
        }
    }
}
```

# 🔐 LOGIN ENDPOINT

```csharp
[HttpPost("/login")]
public async Task<IResult> Login(
    HttpContext ctx,
    UserStore store,
    string username,
    string password)
{
    var user = store.Validate(username, password);

    if (user == null)
        return Results.Unauthorized();

    var claims = new List<Claim>
    {
        new Claim(ClaimTypes.Name, user.UserName)
    };

    foreach (var g in user.Groups)
        claims.Add(new Claim("groups", g));

    var identity = new ClaimsIdentity(claims, "AppCookie");

    await ctx.SignInAsync("AppCookie", new ClaimsPrincipal(identity));

    return Results.Ok();
}
```
