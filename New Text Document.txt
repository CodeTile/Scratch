# 🧱 1. SNAPSHOT MAPPER (ASP.NET LAYER)

```csharp
using System.Security.Claims;

namespace YourApp.Security;

public static class UserSnapshotMapper
{
    public static ExternalUserSnapshot Map(ClaimsPrincipal user)
    {
        return new ExternalUserSnapshot
        {
            UserName = user.Identity?.Name,
            IsAdUser = user.Identity?.AuthenticationType == "Negotiate",

            DirectGroups = user.FindAll("groups")
                .Select(x => x.Value)
                .ToList()
        };
    }
}
```

# 🔐 3. AUTHORIZATION HANDLER

```csharp
using Microsoft.AspNetCore.Authorization;

namespace YourApp.Security;

public class GroupHandler : AuthorizationHandler<GroupRequirement>
{
    private readonly UserContextBuilder _builder;

    public GroupHandler(UserContextBuilder builder)
    {
        _builder = builder;
    }

    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        GroupRequirement requirement)
    {
        var snap = UserSnapshotMapper.Map(context.User);
        var user = _builder.Build(snap);

        if (user.Groups.Any(g =>
            g.Equals(requirement.Group, StringComparison.OrdinalIgnoreCase)))
        {
            context.Succeed(requirement);
        }

        return Task.CompletedTask;
    }
}
```

# ⚙️ 4. EXTENSION METHOD (THE MAIN PIECE)

```csharp
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authentication.Negotiate;
using Microsoft.AspNetCore.Authorization;
using Microsoft.Extensions.DependencyInjection;

namespace YourApp.Security;

public static class SecurityExtensions
{
    public static IServiceCollection AddAppSecurity(
        this IServiceCollection services,
        IConfiguration config)
    {
        // ------------------------
        // AUTHENTICATION
        // ------------------------
        services.AddAuthentication("Hybrid")
            .AddPolicyScheme("Hybrid", "Hybrid Auth", options =>
            {
                options.ForwardDefaultSelector = context =>
                {
                    if (context.User?.Identity?.AuthenticationType == "Negotiate")
                        return NegotiateDefaults.AuthenticationScheme;

                    return "AppCookie";
                };
            })
            .AddNegotiate()
            .AddCookie("AppCookie", options =>
            {
                options.LoginPath = "/login";

                // ONLY affects non-AD users
                options.ExpireTimeSpan = TimeSpan.FromMinutes(7);
                options.SlidingExpiration = true;
            });

        // ------------------------
        // AUTHORIZATION
        // ------------------------
        services.AddAuthorization();

        services.AddSingleton<IAuthorizationHandler, GroupHandler>();

        // ------------------------
        // GROUP CONFIG (CODE-BASED)
        // ------------------------
        var groupMap = new Dictionary<string, List<string>>
        {
            ["IT"] = new() { "Developers" },
            ["Developers"] = new() { "Backend" }
        };

        services.AddSingleton<IGroupResolver>(new GroupResolver(groupMap));

        // ------------------------
        // CORE SERVICES
        // ------------------------
        services.AddScoped<UserContextBuilder>();

        return services;
    }
}
```

# ✅ SIMPLE BLAZOR PAGE / COMPONENT

```razor
@using System.Security.Claims
@using YourApp.Security

@inject AuthenticationStateProvider AuthStateProvider
@inject UserContextBuilder UserContextBuilder

<h3>User Details</h3>

<AuthorizeView>
    <Authorized>
        @if (user == null)
        {
            <p>Loading...</p>
        }
        else
        {
            <p><b>Username:</b> @user.UserName</p>
            <p><b>Email:</b> @email</p>

            <p><b>Groups:</b></p>
            <ul>
                @foreach (var g in user.Groups)
                {
                    <li>@g</li>
                }
            </ul>
        }
    </Authorized>

    <NotAuthorized>
        <p>You are not logged in.</p>
    </NotAuthorized>
</AuthorizeView>

@code {
    private UserContext? user;
    private string? email;

    protected override async Task OnInitializedAsync()
    {
        var authState = await AuthStateProvider.GetAuthenticationStateAsync();
        var principal = authState.User;

        // Build unified user context (groups resolved here)
        var snapshot = UserSnapshotMapper.Map(principal);
        user = UserContextBuilder.Build(snapshot);

        // Email comes from claims (AD or cookie)
        email = principal.FindFirst(ClaimTypes.Email)?.Value
             ?? principal.FindFirst("email")?.Value;
    }
}
```
