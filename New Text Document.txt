# ⚙️ 4. EXTENSION METHOD (THE MAIN PIECE)

```csharp
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authentication.Negotiate;
using Microsoft.AspNetCore.Authorization;
using Microsoft.Extensions.DependencyInjection;

namespace YourApp.Security;

public static class SecurityExtensions
{
    public static IServiceCollection AddAppSecurity(
        this IServiceCollection services,
        IConfiguration config)
    {
        // ------------------------
        // AUTHENTICATION
        // ------------------------
        services.AddAuthentication("Hybrid")
            .AddPolicyScheme("Hybrid", "Hybrid Auth", options =>
            {
                options.ForwardDefaultSelector = context =>
                {
                    if (context.User?.Identity?.AuthenticationType == "Negotiate")
                        return NegotiateDefaults.AuthenticationScheme;

                    return "AppCookie";
                };
            })
            .AddNegotiate()
            .AddCookie("AppCookie", options =>
            {
                options.LoginPath = "/login";

                // ONLY affects non-AD users
                options.ExpireTimeSpan = TimeSpan.FromMinutes(7);
                options.SlidingExpiration = true;
            });

        // ------------------------
        // AUTHORIZATION
        // ------------------------
        services.AddAuthorization();

        services.AddSingleton<IAuthorizationHandler, GroupHandler>();

        // ------------------------
        // GROUP CONFIG (CODE-BASED)
        // ------------------------
        var groupMap = new Dictionary<string, List<string>>
        {
            ["IT"] = new() { "Developers" },
            ["Developers"] = new() { "Backend" }
        };

        services.AddSingleton<IGroupResolver>(new GroupResolver(groupMap));

        // ------------------------
        // CORE SERVICES
        // ------------------------
        services.AddScoped<UserContextBuilder>();

        return services;
    }
}
```

# 🧠 STEP 2 — IMPLEMENTATION (WITH PER-REQUEST CACHE)

We cache inside the request using `HttpContext.Items`.

```csharp id="grp_auth_impl"
using System.Security.Claims;
using Microsoft.AspNetCore.Http;

namespace YourApp.Security;

public class GroupAuthorizer : IGroupAuthorizer
{
    private readonly IHttpContextAccessor _http;
    private readonly UserContextBuilder _builder;

    private const string CacheKey = "UserContext";

    public GroupAuthorizer(
        IHttpContextAccessor http,
        UserContextBuilder builder)
    {
        _http = http;
        _builder = builder;
    }

    public UserContext GetUserContext(ClaimsPrincipal user)
    {
        var ctx = _http.HttpContext;

        if (ctx == null)
        {
            // fallback (non-web scenario)
            var snap = UserSnapshotMapper.Map(user);
            return _builder.Build(snap);
        }

        if (ctx.Items.TryGetValue(CacheKey, out var cached))
            return (UserContext)cached!;

        var snapshot = UserSnapshotMapper.Map(user);
        var result = _builder.Build(snapshot);

        ctx.Items[CacheKey] = result;

        return result;
    }

    public bool IsInGroup(ClaimsPrincipal user, string group)
    {
        var ctx = GetUserContext(user);

        return ctx.Groups.Any(g =>
            g.Equals(group, StringComparison.OrdinalIgnoreCase));
    }
}
```

# 🧠 STEP 2 — DYNAMIC POLICY PROVIDER (IMPORTANT)

Instead of manually registering every group policy, generate them automatically.

```csharp id="policy_provider"
using Microsoft.AspNetCore.Authorization;
using Microsoft.Extensions.Options;

namespace YourApp.Security;

public class GroupPolicyProvider : DefaultAuthorizationPolicyProvider
{
    private const string PREFIX = "GROUP_";

    public GroupPolicyProvider(IOptions<AuthorizationOptions> options)
        : base(options)
    {
    }

    public override Task<AuthorizationPolicy?> GetPolicyAsync(string policyName)
    {
        if (policyName.StartsWith(PREFIX))
        {
            var group = policyName.Substring(PREFIX.Length);

            var policy = new AuthorizationPolicyBuilder()
                .AddRequirements(new GroupRequirement(group))
                .Build();

            return Task.FromResult<AuthorizationPolicy?>(policy);
        }

        return base.GetPolicyAsync(policyName);
    }
}
```

```csharp
    List<string> Expand(List<string> directGroups);
}

public class GroupResolver : IGroupResolver
{
    private readonly IDictionary<string, List<string>> _map;

    public GroupResolver(IDictionary<string, List<string>> map)
    {
        _map = map;
    }

    public List<string> Expand(List<string> directGroups)
    {
        var result = new HashSet<string>();
        var visited = new HashSet<string>();

        foreach (var g in directGroups)
            Resolve(g, result, visited);

        return result.ToList();
    }

    private void Resolve(string g, HashSet<string> result, HashSet<string> visited)
    {
        if (!visited.Add(g)) return;

        result.Add(g);

        if (_map.TryGetValue(g, out var children))
        {
            foreach (var c in children)
                Resolve(c, result, visited);
        }
    }
}
```

# 🧠 GROUP AUTHORIZER (SHARED + CACHED)

```csharp
using System.Security.Claims;
using Microsoft.AspNetCore.Http;

namespace YourApp.Security;

public interface IGroupAuthorizer
{
    bool IsInGroup(ClaimsPrincipal user, string group);
    UserContext GetUserContext(ClaimsPrincipal user);
}

public class GroupAuthorizer : IGroupAuthorizer
{
    private readonly IHttpContextAccessor _http;
    private readonly UserContextBuilder _builder;

    private const string KEY = "UserContext";

    public GroupAuthorizer(IHttpContextAccessor http, UserContextBuilder builder)
    {
        _http = http;
        _builder = builder;
    }

    public UserContext GetUserContext(ClaimsPrincipal user)
    {
        var ctx = _http.HttpContext;

        if (ctx == null)
            return _builder.Build(UserSnapshotMapper.Map(user));

        if (ctx.Items.TryGetValue(KEY, out var cached))
            return (UserContext)cached!;

        var result = _builder.Build(UserSnapshotMapper.Map(user));
        ctx.Items[KEY] = result;

        return result;
    }

    public bool IsInGroup(ClaimsPrincipal user, string group)
    {
        return GetUserContext(user).Groups
            .Any(g => g.Equals(group, StringComparison.OrdinalIgnoreCase));
    }
}
```

# ⚙️ EXTENSION METHOD (MAIN ENTRY POINT)

```csharp
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authentication.Negotiate;
using Microsoft.AspNetCore.Authorization;

namespace YourApp.Security;

public static class SecurityExtensions
{
    public static IServiceCollection AddAppSecurity(this IServiceCollection services)
    {
        // AUTH
        services.AddAuthentication("Hybrid")
            .AddPolicyScheme("Hybrid", "Hybrid", options =>
            {
                options.ForwardDefaultSelector = ctx =>
                {
                    if (ctx.User?.Identity?.AuthenticationType == "Negotiate")
                        return NegotiateDefaults.AuthenticationScheme;

                    return "AppCookie";
                };
            })
            .AddNegotiate()
            .AddCookie("AppCookie", o =>
            {
                o.ExpireTimeSpan = TimeSpan.FromMinutes(7);
                o.SlidingExpiration = true;
            });

        services.AddAuthorization();

        // CORE
        var groupMap = new Dictionary<string, List<string>>
        {
            ["IT"] = new() { "Developers" },
            ["Developers"] = new() { "Backend" }
        };

        services.AddSingleton<IGroupResolver>(new GroupResolver(groupMap));
        services.AddScoped<UserContextBuilder>();

        // AUTHZ
        services.AddHttpContextAccessor();
        services.AddScoped<IGroupAuthorizer, GroupAuthorizer>();
        services.AddScoped<IAuthorizationHandler, GroupHandler>();
        services.AddSingleton<IAuthorizationPolicyProvider, GroupPolicyProvider>();

        return services;
    }
}
```

```razor
@using Microsoft.AspNetCore.Authorization
@inject IAuthorizationService AuthService
@inject AuthenticationStateProvider AuthStateProvider

@if (allowed)

    [Parameter] public string Group { get; set; } = "";

    protected override async Task OnParametersSetAsync()
    {
        var state = await AuthStateProvider.GetAuthenticationStateAsync();
        var policy = GroupAuthorizeAttribute.Build(Group);

        var result = await AuthService.AuthorizeAsync(state.User, policy);

        allowed = result.Succeeded;
    }
}
```
