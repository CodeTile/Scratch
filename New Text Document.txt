```yaml id="s0s3zr"
trigger:
- main

pool:
  vmImage: 'windows-latest'

variables:
  BuildConfiguration: 'Release'
  SolutionFile: 'FooBar.slnx'

steps:

# ============================================================
# 1. Clean sources
# ============================================================
- checkout: self
  clean: true

# ============================================================
# Display tool versions used by the build agent
#
# Displays:
#   - Git version
#   - NuGet version
#   - .NET SDK version
#   - Full .NET SDK information
#
# This pipeline assumes:
#   - NuGet is already available on the build agent
#   - .NET 10 SDK is already installed on the build agent
# ============================================================
- powershell: |
    Write-Host "========================================"
    Write-Host "Git Version"
    Write-Host "========================================"
    git --version

    Write-Host ""
    Write-Host "========================================"
    Write-Host "NuGet Version"
    Write-Host "========================================"
    nuget help | Select-String "NuGet Version"

    Write-Host ""
    Write-Host "========================================"
    Write-Host ".NET SDK Version"
    Write-Host "========================================"
    dotnet --version

    Write-Host ""
    Write-Host "========================================"
    Write-Host ".NET SDK Details"
    Write-Host "========================================"
    dotnet --info
  displayName: 'Display Tool Versions'

# ============================================================
# 2. Restore NuGet packages
# ============================================================
- task: NuGetCommand@2
  displayName: 'Restore NuGet Packages'
  inputs:
    command: 'restore'
    restoreSolution: '$(SolutionFile)'

# ============================================================
# 3. Build solution
# ============================================================
- task: DotNetCoreCLI@2
  displayName: 'Build Solution'
  inputs:
    command: 'build'
    projects: '$(SolutionFile)'
    arguments: '--configuration $(BuildConfiguration) --no-restore'

# ============================================================
# Install ReportGenerator tool
# ============================================================
- script: |
    dotnet tool install -g dotnet-reportgenerator-globaltool
  displayName: 'Install ReportGenerator'

# ============================================================
# 4. Run all tests with code coverage
#
# IMPORTANT:
# Test projects must reference the NuGet package:
#
#   <PackageReference Include="coverlet.collector" Version="6.*">
#     <PrivateAssets>all</PrivateAssets>
#   </PackageReference>
#
# Coverage exclusions:
#   [*Tests*]*
#     Excludes any assembly/project containing "Tests"
#
# Coverage is collected using:
#   --collect:"XPlat Code Coverage"
# ============================================================
- task: DotNetCoreCLI@2
  displayName: 'Run Tests with Coverage'
  inputs:
    command: 'test'
    projects: '$(SolutionFile)'
    arguments: >
      --configuration $(BuildConfiguration)
      --no-build
      --collect:"XPlat Code Coverage"
      --settings .runsettings
      --logger trx
    publishTestResults: true

# ============================================================
# Generate HTML + Cobertura coverage report
#
# Coverage report files generated:
#
#   $(Build.SourcesDirectory)/CoverageReport/index.html
#   $(Build.SourcesDirectory)/CoverageReport/Cobertura.xml
#
# Html:
#   Generates a browsable HTML report.
#
# Cobertura:
#   Generates XML format consumed by Azure DevOps.
# ============================================================
- script: |
    reportgenerator ^
      -reports:"$(Agent.TempDirectory)/**/coverage.cobertura.xml" ^
      -targetdir:"$(Build.SourcesDirectory)/CoverageReport" ^
      -reporttypes:"Html;Cobertura"
  displayName: 'Generate Coverage Report'

# ============================================================
# Publish coverage results to Azure DevOps
#
# HOW TO VIEW COVERAGE:
#
# 1. Open Azure DevOps
# 2. Go to Pipelines
# 3. Open the completed pipeline run
# 4. Select the "Code Coverage" tab
#
# Azure DevOps reads coverage data from:
#   CoverageReport/Cobertura.xml
# ============================================================
- task: PublishCodeCoverageResults@2
  displayName: 'Publish Code Coverage'
  inputs:
    summaryFileLocation: '$(Build.SourcesDirectory)/CoverageReport/Cobertura.xml'
    pathToSources: '$(Build.SourcesDirectory)'
    failIfCoverageEmpty: true

# ============================================================
# Publish full HTML report as downloadable artifact
#
# HOW TO VIEW HTML REPORT:
#
# 1. Open the completed pipeline run
# 2. Open the "Artifacts" section
# 3. Download artifact: CodeCoverageReport
# 4. Open:
#      index.html
#
# This provides detailed per-file and per-line coverage.
# ============================================================
- task: PublishPipelineArtifact@1
  displayName: 'Publish Coverage HTML Report'
  inputs:
    targetPath: '$(Build.SourcesDirectory)/CoverageReport'
    artifact: 'CodeCoverageReport'

# ============================================================
# Tag successful build with the Azure DevOps build number
#
# Example tag:
#   20260515.3
#
# HOW TO VIEW TAG:
#
# 1. Open Pipelines
# 2. Open the completed build
# 3. View "Tags" section
#
# Build.BuildNumber is automatically generated by Azure DevOps.
# ============================================================
- powershell: |
    Write-Host "Tagging build with:"
    Write-Host "$(Build.BuildNumber)"

    Write-Host "##vso[build.addbuildtag]$(Build.BuildNumber)"
  displayName: 'Tag Build With Build Number'
  condition: succeeded()
```

<RunSettings>

  <RunConfiguration>
    <ResultsDirectory>TestResults</ResultsDirectory>
    <TargetPlatform>x64</TargetPlatform>
    <MaxCpuCount>0</MaxCpuCount>
  </RunConfiguration>

  <DataCollectionRunSettings>

    <DataCollectors>

      <DataCollector friendlyName="XPlat code coverage">

        <Configuration>

          <!--
            Coverage output format.
            Cobertura works well with Azure DevOps.
          -->
          <Format>cobertura</Format>

          <!--
            ====================================================
            EXCLUDE ASSEMBLIES
            ====================================================

            Excludes:
              - All test projects
              - All Microsoft assemblies
              - Common third-party test libraries
          -->
          <Exclude>

            [*Tests*]*,

            [Microsoft.*]*,

            [xunit*]*,
            [FluentAssertions*]*,
            [Moq*]*,
            [coverlet*]*

          </Exclude>

          <!--
            ====================================================
            EXCLUDE FILES
            ====================================================

            Excludes:
              - Program.cs
              - generated files
              - obj/bin folders
              - migrations
          -->
          <ExcludeByFile>

            **/Program.cs,

            **/bin/**,

            **/obj/**,

            **/*.g.cs,

            **/*.generated.cs,

            **/*Designer.cs,

            **/Migrations/**

          </ExcludeByFile>

          <!--
            ====================================================
            EXCLUDE ATTRIBUTES
            ====================================================
          -->
          <ExcludeByAttribute>

            Obsolete,

            GeneratedCodeAttribute,

            CompilerGeneratedAttribute,

            ExcludeFromCodeCoverageAttribute

          </ExcludeByAttribute>

          <!--
            Recommended defaults
          -->
          <IncludeTestAssembly>false</IncludeTestAssembly>

          <SingleHit>false</SingleHit>

          <SkipAutoProps>true</SkipAutoProps>

          <DeterministicReport>true</DeterministicReport>

          <UseSourceLink>true</UseSourceLink>

        </Configuration>

      </DataCollector>

    </DataCollectors>

  </DataCollectionRunSettings>

</RunSettings>
