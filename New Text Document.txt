## 📌 SecurityHelpers.cs

```csharp
namespace Security.Helpers;

public static class SecurityHelpers
{
    public static (string Domain, string User) ParseWindowsIdentity(string identity)
    {
        var parts = identity.Split('\\');
        return (parts[0], parts[1]);
    }

    public static bool ValidateWindowsDomain(HttpContext context, SecurityConfig config)
    {
        var name = context.User.Identity?.Name;
        if (string.IsNullOrWhiteSpace(name)) return false;

        var (domain, _) = ParseWindowsIdentity(name);
        return domain.Equals(config.AllowedDomain, StringComparison.OrdinalIgnoreCase);
    }

    public static UserRole ResolveRole(IEnumerable<string> userGroups, RoleMappingConfig config)
    {
        foreach (var kvp in config.RoleMappings)
        {
            var roleName = kvp.Key;
            var adGroups = kvp.Value;

            if (userGroups.Any(g => adGroups.Contains(g, StringComparer.OrdinalIgnoreCase)))
            {
                return Enum.Parse<UserRole>(roleName);
            }
        }

        return UserRole.None;
    }
}
```

```csharp
namespace Security.Caching;

public class UserRoleCache : IUserRoleCache
{
    private readonly MemoryCache _cache = new(new MemoryCacheOptions());
    private readonly SecurityConfig _security;

    public UserRoleCache(IOptions<SecurityConfig> security)
    {
        _security = security.Value;
    }

    public bool TryGet(string userKey, out UserRole role)
    {
        return _cache.TryGetValue(userKey, out role);
    }

    public void Set(string userKey, UserRole role)
    {
        var minutes = _security.Cache.UserRoleCacheMinutes;

        _cache.Set(
            userKey,
            role,
            TimeSpan.FromMinutes(minutes)
        );
    }
}
```

## 📌 SecurityMiddleware.cs

```csharp
namespace Security.Middleware;

public class SecurityMiddleware
{
    private readonly RequestDelegate _next;

    public SecurityMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task Invoke(
        HttpContext context,
        IOptions<SecurityConfig> security,
        IUserRoleCache cache)
    {
        var sec = security.Value;

        // 1. Domain validation
        if (context.User.Identity?.IsAuthenticated == true)
        {
            if (!SecurityHelpers.ValidateWindowsDomain(context, sec))
            {
                context.Response.StatusCode = StatusCodes.Status403Forbidden;
                return;
            }
        }

        // 2. Cache key
        var userKey = context.User.Identity?.Name ?? "anonymous";

        // 3. Try cache
        if (!cache.TryGet(userKey, out var userRole))
        {
            var groups = context.User.Claims
                .Where(c => c.Type == ClaimTypes.GroupSid)
                .Select(c => new SecurityIdentifier(c.Value)
                    .Translate(typeof(NTAccount)).Value);

            userRole = SecurityHelpers.ResolveRole(groups, sec.RoleMappings);

            cache.Set(userKey, userRole);
        }

        // 4. Check endpoint requirement
        var endpoint = context.GetEndpoint();
        var attr = endpoint?.Metadata.GetMetadata<RequireRoleAttribute>();

        if (attr != null)
        {
            if ((int)userRole < (int)attr.MinimumRole)
            {
                context.Response.StatusCode = StatusCodes.Status403Forbidden;
                return;
            }
        }

        await _next(context);
    }
}
```

```csharp
public class FakeIdentityMiddleware
{
    private readonly RequestDelegate _next;

    public FakeIdentityMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task Invoke(HttpContext ctx)
    {
        var domain = ctx.Request.Headers["X-Domain"].FirstOrDefault();
        var user = ctx.Request.Headers["X-User"].FirstOrDefault();
        var groups = ctx.Request.Headers["X-Group"];

        if (domain != null && user != null)
        {
            var claims = new List<Claim>
            {
                new Claim(ClaimTypes.Name, $"{domain}\\{user}")
            };

            foreach (var g in groups)
            {
                var sid = new SecurityIdentifier("S-1-5-21-FAKE");
                claims.Add(new Claim(ClaimTypes.GroupSid, sid.Value));
            }

            ctx.User = new ClaimsPrincipal(new ClaimsIdentity(claims, "TestAuth"));
        }

        await _next(ctx);
    }
}
```

```csharp
public class ActiveDirectoryService : IActiveDirectoryService
{

    private readonly SecurityConfig _config;

    public ActiveDirectoryService(IOptions<SecurityConfig> config)
    {
        _config = config.Value;
    }

    public async Task<UserRole> GetUserRoleAsync(string domain, string username)
    {
        var cacheKey = $"{domain}\\{username}";

        if (_cache.TryGetValue(cacheKey, out UserRole cachedRole))

        // 🔥 Your AD interrogation logic here
        var groups = await QueryActiveDirectoryAsync(domain, username);

        var role = ResolveRole(groups);

        _cache.Set(cacheKey, role, TimeSpan.FromMinutes(_config.Cache.UserRoleCacheMinutes));
```

# 🧩 2. Middleware calls your AD class

Middleware becomes **very small**:

```csharp
public class SecurityMiddleware
{
    private readonly RequestDelegate _next;

    public SecurityMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task Invoke(
        HttpContext context,
        IActiveDirectoryService adService,
        IOptions<SecurityConfig> security)
    {
        var sec = security.Value;

        // Extract identity
        var identity = context.User.Identity?.Name;

        if (identity is null)
        {
            context.Response.StatusCode = StatusCodes.Status401Unauthorized;
            return;
        }

        var (domain, username) = SecurityHelpers.ParseWindowsIdentity(identity);

        // Domain enforcement
        if (!domain.Equals(sec.AllowedDomain, StringComparison.OrdinalIgnoreCase))
        {
            context.Response.StatusCode = StatusCodes.Status403Forbidden;
            return;
        }

        // 🔥 Get role from your AD class (cached)
        var userRole = await adService.GetUserRoleAsync(domain, username);

        // Check endpoint requirement
        var endpoint = context.GetEndpoint();
        var attr = endpoint?.Metadata.GetMetadata<RequireRoleAttribute>();

        if (attr != null)
        {
            if ((int)userRole < (int)attr.MinimumRole)
            {
                context.Response.StatusCode = StatusCodes.Status403Forbidden;
                return;
            }
        }

        await _next(context);
    }
}
```
