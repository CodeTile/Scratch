using Microsoft.Extensions.DependencyInjection;
using Polly;
using Polly.Registry;
using Polly.Timeout;
using Shouldly;
using Xunit;

public class ResiliencePipelineTests
{
    [Fact]
    public async Task All_resilience_pipelines_should_resolve_and_timeout()
    {
        // Arrange
        var services = new ServiceCollection();

        services.AddLogging();

        services.AddResiliencePipeline("fast-timeout", builder =>
        {
            builder.AddTimeout(TimeSpan.FromMilliseconds(100));
        });

        services.AddResiliencePipeline("slow-timeout", builder =>
        {
            builder.AddTimeout(TimeSpan.FromMilliseconds(200));
        });

        using var provider = services.BuildServiceProvider(
            new ServiceProviderOptions
            {
                ValidateScopes = true,
                ValidateOnBuild = true
            });

        using var scope = provider.CreateScope();

        var pipelineProvider =
            scope.ServiceProvider
                .GetRequiredService<ResiliencePipelineProvider<string>>();

        pipelineProvider.ShouldNotBeNull();

        //
        // Find all named resilience pipelines registered in IServiceCollection
        //
        var pipelineNames = services
            .Where(x =>
                x.ServiceType == typeof(ResiliencePipelineRegistry<string>))
            .ToList();

        //
        // Since Polly named pipelines are not directly exposed from IServiceCollection,
        // keep your known pipeline names centrally here.
        //
        var knownPipelineNames = new[]
        {
            "fast-timeout",
            "slow-timeout"
        };

        foreach (var pipelineName in knownPipelineNames)
        {
            //
            // Resolve pipeline
            //
            var pipeline = pipelineProvider.GetPipeline(pipelineName);

            pipeline.ShouldNotBeNull();

            //
            // Verify timeout strategy actually works
            //
            await Should.ThrowAsync<TimeoutRejectedException>(
                async () =>
                {
                    await pipeline.ExecuteAsync(async token =>
                    {
                        await Task.Delay(TimeSpan.FromSeconds(5), token);
                    });
                });
        }
    }
}
