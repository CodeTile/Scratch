## `AdClaimsTransformation.cs`

```csharp
using Microsoft.AspNetCore.Authentication;
using Microsoft.Extensions.Options;
using System.Security.Claims;

public class AdClaimsTransformation
    : IClaimsTransformation
{
    private readonly IAdService _adService;
    private readonly AuthOptions _options;

    public AdClaimsTransformation(
        IAdService adService,
        IOptions<AuthOptions> options)
    {
        _adService = adService;
        _options = options.Value;
    }

    public Task<ClaimsPrincipal> TransformAsync(
        ClaimsPrincipal principal)
    {
        if (principal.Identity is not ClaimsIdentity identity)
        {
            return Task.FromResult(principal);
        }

        if (!identity.IsAuthenticated)
        {
            return Task.FromResult(principal);
        }

        // Prevent duplicate claims
        if (identity.Claims.Any(c => c.Type == ClaimTypes.Role))
        {
            return Task.FromResult(principal);
        }

        var username = identity.Name;

        if (string.IsNullOrWhiteSpace(username))
        {
            return Task.FromResult(principal);
        }

        var userGroups =
            _adService.GetUserGroups(username);

        var roles = new HashSet<string>(
            StringComparer.OrdinalIgnoreCase);

        foreach (var mapping in _options.Groups)
        {
            var adGroup = mapping.Key;
            var mappedRoles = mapping.Value;

            if (userGroups.Contains(
                    adGroup,
                    StringComparer.OrdinalIgnoreCase))
            {
                foreach (var role in mappedRoles)
                {
                    roles.Add(role);
                }
            }
        }

        foreach (var role in roles)
        {
            identity.AddClaim(
                new Claim(
                    ClaimTypes.Role,
                    role));
        }

        return Task.FromResult(principal);
    }
}
```

That’s a good approach. In enterprise apps it’s common to move authentication/authorization registration into extension methods or middleware setup classes to keep `Program.cs` clean.

A typical structure would look like this:

# Program.cs

```csharp id="9x0s9v"
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddApplicationAuthentication(
    builder.Configuration);

builder.Services.AddApplicationAuthorization(
    builder.Configuration);

builder.Services.AddRazorComponents()
    .AddInteractiveServerComponents();

var app = builder.Build();

app.UseApplicationSecurity();

app.MapRazorComponents<App>();

app.Run();
```

# Service Registration Extensions

## `AuthenticationExtensions.cs`

```csharp id="zbyr9j"
using Microsoft.AspNetCore.Authentication.Negotiate;

public static class AuthenticationExtensions
{
    public static IServiceCollection
        AddApplicationAuthentication(
            this IServiceCollection services,
            IConfiguration configuration)
    {
        services
            .AddAuthentication(
                NegotiateDefaults.AuthenticationScheme)
            .AddNegotiate();

        services.AddAuthorization();

        services.Configure<AuthOptions>(
            configuration.GetSection("Auth"));

        services.AddScoped<IAdService, AdService>();

        services.AddTransient<IClaimsTransformation,
            AdClaimsTransformation>();

        return services;
    }
}
```

| --------------------- | ----------------------- |
| Windows Auth          | User authentication     |
| IClaimsTransformation | AD group → role mapping |
| Config                | Authorization rules     |
| Blazor                | UI enforcement          |

That keeps the system clean, testable, and maintainable.
