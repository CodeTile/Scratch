using Microsoft.EntityFrameworkCore;
using Shouldly;
using Xunit;
using System.Threading;

public class CancellationTokenTests
{
    private TestDbContext CreateContext()
    {
        var options = new DbContextOptionsBuilder<TestDbContext>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString())
            .Options;

        return new TestDbContext(options);
    }

    // ------------------------------------------------------------
    // AddAsync
    // ------------------------------------------------------------
    [Fact]
    public async Task AddAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var entity = new FooBar { Id = 1, Name = "A" };
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.AddAsync(entity, cts.Token));
    }

    // ------------------------------------------------------------
    // AddBatchAsync
    // ------------------------------------------------------------
    [Fact]
    public async Task AddBatchAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var entities = new[]
        {
            new FooBar { Id = 1, Name = "A" },
            new FooBar { Id = 2, Name = "B" }
        };
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.AddBatchAsync(entities, cts.Token));
    }

    // ------------------------------------------------------------
    // UpdateAsync (single)
    // ------------------------------------------------------------
    [Fact]
    public async Task UpdateAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.Add(new FooBar { Id = 1, Name = "Old" });
        await ctx.SaveChangesAsync();
        ctx.ChangeTracker.Clear();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var updated = new FooBar { Id = 1, Name = "New" };
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.UpdateAsync(updated, cts.Token));
    }

    // ------------------------------------------------------------
    // UpdateAsync (batch)
    // ------------------------------------------------------------
    [Fact]
    public async Task UpdateBatchAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.AddRange(
            new FooBar { Id = 1, Name = "A" },
            new FooBar { Id = 2, Name = "B" }
        );
        await ctx.SaveChangesAsync();
        ctx.ChangeTracker.Clear();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var updated = new[]
        {
            new FooBar { Id = 1, Name = "X" },
            new FooBar { Id = 2, Name = "Y" }
        };
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.UpdateAsync(updated, cts.Token));
    }

    // ------------------------------------------------------------
    // DeleteAsync (predicate)
    // ------------------------------------------------------------
    [Fact]
    public async Task DeleteAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.Add(new FooBar { Id = 1 });
        await ctx.SaveChangesAsync();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.DeleteAsync(e => e.Id == 1, cts.Token));
    }

    // ------------------------------------------------------------
    // DeleteAsync (entity)
    // ------------------------------------------------------------
    [Fact]
    public async Task DeleteEntityAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        var entity = new FooBar { Id = 1 };
        ctx.FooBars.Add(entity);
        await ctx.SaveChangesAsync();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.DeleteAsync(entity, cts.Token));
    }

    // ------------------------------------------------------------
    // GetByIdAsync
    // ------------------------------------------------------------
    [Fact]
    public async Task GetByIdAsync_ShouldCancel_WhenTokenCancelled()
    {
        // Arrange
        var ctx = CreateContext();
        ctx.FooBars.Add(new FooBar { Id = 1 });
        await ctx.SaveChangesAsync();
        var repo = new GenericCrudRepository<FooBar>(ctx);
        var cts = new CancellationTokenSource();
        cts.Cancel();

        // Act + Assert
        await Should.ThrowAsync<TaskCanceledException>(() =>
            repo.GetByIdAsync(e => e.Id == 1, cts.Token));
    }
}
