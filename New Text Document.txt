# ⚡ 4. AD CACHE SERVICE (7 min TTL support)

```csharp id="cache"
using Microsoft.Extensions.Caching.Memory;

namespace YourApp.Security;

public interface IAdCache
{
    T? GetOrCreate<T>(string key, Func<T> factory);
}

public class AdCache : IAdCache
{
    private readonly IMemoryCache _cache;

    public AdCache(IMemoryCache cache)
    {
        _cache = cache;
    }

    public T? GetOrCreate<T>(string key, Func<T> factory)
    {
        return _cache.GetOrCreate(key, entry =>
        {
            entry.SlidingExpiration = TimeSpan.FromMinutes(7);
            entry.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10);

            return factory();
        });
    }
}
```

# 🧠 5. GROUP RESOLUTION (supports nested groups)

```csharp id="groups"
namespace YourApp.Security;

public interface IGroupResolver
{
    List<string> ExpandGroups(List<string> directGroups);
}

public class GroupResolver : IGroupResolver
{
    private readonly IDictionary<string, List<string>> _groupMap;

    public GroupResolver(IDictionary<string, List<string>> groupMap)
    {
        _groupMap = groupMap;
    }

    public List<string> ExpandGroups(List<string> directGroups)
    {
        var result = new HashSet<string>();
        var visited = new HashSet<string>();

        foreach (var g in directGroups)
            Resolve(g, result, visited);

        return result.ToList();
    }

    private void Resolve(string group, HashSet<string> result, HashSet<string> visited)
    {
        if (!visited.Add(group))
            return;

        result.Add(group);

        if (_groupMap.TryGetValue(group, out var children))
        {
            foreach (var c in children)
                Resolve(c, result, visited);
        }
    }
}
```

# 🧠 6. GROUP → ROLE MAPPING (CONFIGURED IN CODE)

```csharp id="roles"
namespace YourApp.Security;

public interface IRoleMapper
{
    List<string> Map(List<string> groups);
}

public class RoleMapper : IRoleMapper
{
    private readonly IDictionary<string, string> _map;

    public RoleMapper(IDictionary<string, string> map)
    {
        _map = map;
    }

    public List<string> Map(List<string> groups)
    {
        return groups
            .Where(g => _map.ContainsKey(g))
            .Select(g => _map[g])
            .Distinct()
            .ToList();
    }
}
```

# 🧠 7. CORE USER CONTEXT BUILDER

```csharp id="core"
namespace YourApp.Security;

public class UserContextBuilder
{
    private readonly IGroupResolver _groups;
    private readonly IRoleMapper _roles;

    public UserContextBuilder(IGroupResolver groups, IRoleMapper roles)
    {
        _groups = groups;
        _roles = roles;
    }

    public UserContext Build(ExternalUserSnapshot input)
    {
        var allGroups = _groups.ExpandGroups(input.DirectGroups);
        var roles = _roles.Map(allGroups);

        return new UserContext
        {
            UserName = input.UserName,
            Groups = allGroups,
            Roles = roles,
            IsAdUser = input.IsAdUser
        };
    }
}
```

# 🌐 8. HOST LAYER (Blazor/API mapping only)

## Claims → Snapshot

```csharp id="map"
using System.Security.Claims;
using YourApp.Security;

public static class UserSnapshotMapper
{
    public static ExternalUserSnapshot ToSnapshot(ClaimsPrincipal user)
    {
        return new ExternalUserSnapshot
        {
            UserName = user.Identity?.Name,
            IsAdUser = user.Identity?.AuthenticationType == "Negotiate",

            DirectGroups = user.FindAll("groups")
                .Select(x => x.Value)
                .ToList()
        };
    }
}
```
