[Fact]
public async Task PatchAsync_Generic_RetriesAndSucceeds()
{
    // Arrange
    var client = CreateClient(call =>
    {
        if (call < 3)
        {
            throw new HttpRequestException();
        }

        return new HttpResponseMessage(HttpStatusCode.OK)
        {
            Content = JsonContent.Create(new { Message = "Patched" })
        };
    });

    var sut = CreateSut(client);

    // Act
    var result = await sut.PatchAsync<object, dynamic>("Foo", new { Name = "OnTheNingNangNong" });

    // Assert
    result.ShouldNotBeNull();

}

[Fact]
public async Task PatchAsync_Generic_DoesNotRetryOnBadRequest()
{
    // Arrange
    int callCount = 0;

    var client = CreateClient(call =>
    {
        callCount++;

        if (callCount == 1)
        {
            return new HttpResponseMessage(HttpStatusCode.BadRequest);
        }

        return new HttpResponseMessage(HttpStatusCode.OK);
    });

    var sut = CreateSut(client);

    // Act
    var result = await sut.PatchAsync<object, dynamic>("Foo", new { Name = "OnTheNingNangNong" });

    // Assert
    result.ShouldBeNull();
    callCount.ShouldBe(1);
}

[Fact]
public async Task PatchAsync_Generic_CallsCorrectUrlAndMethod()
{
    // Arrange
    HttpRequestMessage? captured = null;

    var client = CreateClient(
        _ =>
        {
            return new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = JsonContent.Create(new { Message = "OK" })
            };
        },
        (req, _) =>
        {
            captured = req;
        }
    );

    var sut = CreateSut(client);

    // Act
    await sut.PatchAsync<object, dynamic>("Foo", new { Name = "OnTheNingNangNong" });

    // Assert
    captured.ShouldNotBeNull();
    captured!.RequestUri!.ToString().ShouldBe("Foo");
    captured.Method.Method.ShouldBe("PATCH");
}

[Fact]
public async Task PatchAsync_Generic_SendsPayloadCorrectly()
{
    // Arrange
    HttpRequestMessage? captured = null;

    var client = CreateClient(
        _ =>
        {
            return new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = JsonContent.Create(new { Message = "OK" })
            };
        },
        (req, _) =>
        {
            captured = req;
        }
    );

    var sut = CreateSut(client);

    var payload = new { Name = "OnTheNingNangNong" };

    // Act
    await sut.PatchAsync<object, dynamic>("Foo", payload);

    // Assert
    captured.ShouldNotBeNull();

    var sentJson = await captured!.Content!.ReadAsStringAsync();
    sentJson.ShouldContain("OnTheNingNangNong");
}

[Fact]
public async Task PatchAsync_Generic_RespectsCancellationToken()
{
    // Arrange
    using var cts = new CancellationTokenSource();
    cts.Cancel();

    var client = CreateClient(_ =>
    {
        return new HttpResponseMessage(HttpStatusCode.OK);
    });

    var sut = CreateSut(client);

    // Act & Assert
    await Should.ThrowAsync<TaskCanceledException>(async () =>
        await sut.PatchAsync<object, dynamic>("Foo", new { Name = "OnTheNingNangNong" }, cts.Token)
    );
}
