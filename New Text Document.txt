using System.Net;
using System.Net.Http.Json;
using Microsoft.Extensions.Options;
using Polly;
using Polly.Retry;
using Polly.Timeout;

namespace MyApp;

public interface IMyClass
{
    Task<string?> GetFooAsync(CancellationToken cancellationToken = default);
    Task<string?> GetBarAsync(CancellationToken cancellationToken = default);

    Task<string?> PostFooAsync(object payload, CancellationToken cancellationToken = default);
    Task<TResponse?> PostBarAsync<TRequest, TResponse>(TRequest payload, CancellationToken cancellationToken = default);

    Task<string?> PutAsync(string url, object payload, CancellationToken cancellationToken = default);
    Task<TResponse?> PutAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default);

    Task<string?> PatchAsync(string url, object payload, CancellationToken cancellationToken = default);
    Task<TResponse?> PatchAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default);

    Task<bool> DeleteAsync(string url, CancellationToken cancellationToken = default);
}

public class MyClass : IMyClass
{
    private readonly HttpClient _httpClient;
    private readonly ResiliencePipeline<HttpResponseMessage> _pipeline;

    public MyClass(IHttpClientFactory httpClientFactory, IOptions<PollySettings> settings)
    {
        _httpClient = httpClientFactory.CreateClient("Hello");
        _pipeline = BuildPipeline(settings.Value);
    }

    // ---------------- PUBLIC API ----------------

    public Task<string?> GetFooAsync(CancellationToken cancellationToken = default)
        => GetResultAsStringAsync("Foo", cancellationToken);

    public Task<string?> GetBarAsync(CancellationToken cancellationToken = default)
        => GetResultAsync<string>("Bar", cancellationToken);

    public Task<string?> PostFooAsync(object payload, CancellationToken cancellationToken = default)
        => PostResultAsStringAsync("Foo", payload, cancellationToken);

    public Task<TResponse?> PostBarAsync<TRequest, TResponse>(TRequest payload, CancellationToken cancellationToken = default)
        => PostResultAsync<TRequest, TResponse>("Bar", payload, cancellationToken);

    public Task<string?> PutAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PutResultAsStringAsync(url, payload, cancellationToken);

    public Task<TResponse?> PutAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PutResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<string?> PatchAsync(string url, object payload, CancellationToken cancellationToken = default)
        => PatchResultAsStringAsync(url, payload, cancellationToken);

    public Task<TResponse?> PatchAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken = default)
        => PatchResultAsync<TRequest, TResponse>(url, payload, cancellationToken);

    public Task<bool> DeleteAsync(string url, CancellationToken cancellationToken = default)
        => DeleteResultAsync(url, cancellationToken);

    // ---------------- GET ----------------

    private async Task<string?> GetResultAsStringAsync(string url, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.GetAsync(url, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<T?> GetResultAsync<T>(string url, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.GetAsync(url, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<T>(cancellationToken: cancellationToken);
    }

    // ---------------- POST ----------------

    private async Task<string?> PostResultAsStringAsync(string url, object payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.PostAsJsonAsync(url, payload, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PostResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.PostAsJsonAsync(url, payload, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- PUT ----------------

    private async Task<string?> PutResultAsStringAsync(string url, object payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.PutAsJsonAsync(url, payload, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PutResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.PutAsJsonAsync(url, payload, ct), cancellationToken);

        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- PATCH ----------------

    private async Task<string?> PatchResultAsStringAsync(string url, object payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(
            async ct =>
            {
                using var request = new HttpRequestMessage(HttpMethod.Patch, url)
                {
                    Content = JsonContent.Create(payload)
                };
                return await _httpClient.SendAsync(request, ct);
            },
            cancellationToken);

        if (!response.IsSuccessStatusCode)
            return null;

        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    private async Task<TResponse?> PatchResultAsync<TRequest, TResponse>(string url, TRequest payload, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(
            async ct =>
            {
                using var request = new HttpRequestMessage(HttpMethod.Patch, url)
                {
                    Content = JsonContent.Create(payload)
                };
                return await _httpClient.SendAsync(request, ct);
            },
            cancellationToken);

        if (!response.IsSuccessStatusCode)
            return default;

        return await response.Content.ReadFromJsonAsync<TResponse>(cancellationToken: cancellationToken);
    }

    // ---------------- DELETE ----------------

    private async Task<bool> DeleteResultAsync(string url, CancellationToken cancellationToken)
    {
        var response = await ExecuteAsync(ct => _httpClient.DeleteAsync(url, ct), cancellationToken);
        return response.IsSuccessStatusCode;
    }

    // ---------------- PIPELINE EXECUTION ----------------

    private Task<HttpResponseMessage> ExecuteAsync(
        Func<CancellationToken, Task<HttpResponseMessage>> action,
        CancellationToken cancellationToken)
        => _pipeline.ExecuteAsync(
            static async (context, ct) =>
            {
                var func = context.GetValue<Func<CancellationToken, Task<HttpResponseMessage>>>("action");
                return await func(ct);
            },
            new ResilienceContext
            {
                CancellationToken = cancellationToken
            }.WithValue("action", action));

    // ---------------- PIPELINE DEFINITION ----------------

    private static ResiliencePipeline<HttpResponseMessage> BuildPipeline(PollySettings settings)
    {
        var retryOptions = new RetryStrategyOptions<HttpResponseMessage>
        {
            MaxRetryAttempts = settings.RetryCount,
            Delay = TimeSpan.FromMilliseconds(settings.RetryBaseDelayMs),
            BackoffType = DelayBackoffType.Exponential,
            UseJitter = true,
            ShouldHandle = args =>
            {
                var outcome = args.Outcome;

                if (outcome.Exception is HttpRequestException)
                    return ValueTask.FromResult(true);

                if (outcome.Exception is TaskCanceledException &&
                    !args.Context.CancellationToken.IsCancellationRequested)
                    return ValueTask.FromResult(true);

                if (outcome.Result is { } response && IsTransientStatus(response.StatusCode))
                    return ValueTask.FromResult(true);

                return ValueTask.FromResult(false);
            }
        };

        var timeoutOptions = new TimeoutStrategyOptions<HttpResponseMessage>
        {
            Timeout = TimeSpan.FromSeconds(settings.TimeoutSeconds)
        };

        return new ResiliencePipelineBuilder<HttpResponseMessage>()
            .AddTimeout(timeoutOptions)
            .AddRetry(retryOptions)
            .Build();
    }

    private static bool IsTransientStatus(HttpStatusCode statusCode)
        => statusCode is HttpStatusCode.RequestTimeout      // 408 - request timed out
        or (HttpStatusCode)429                              // 429 - too many requests (rate limited)
        or HttpStatusCode.InternalServerError               // 500 - server error
        or HttpStatusCode.BadGateway                        // 502 - bad gateway
        or HttpStatusCode.ServiceUnavailable                // 503 - service unavailable
        or HttpStatusCode.GatewayTimeout;                   // 504 - upstream gateway timeout
}

file static class ResilienceContextExtensions
{
    public static ResilienceContext WithValue<T>(this ResilienceContext context, string key, T value)
    {
        context.Properties.Set(new ResiliencePropertyKey<T>(key), value);
        return context;
    }

    public static T GetValue<T>(this ResilienceContext context, string key)
        => context.Properties.GetValue(new ResiliencePropertyKey<T>(key));
